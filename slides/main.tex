\section{Problem}

\begin{frame}{Under load, Linux delivers results late, and the receiver cannot tell}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item Linux: high throughput, \alert{no timing guarantee}
        \item Late results look exactly like fresh ones
        \item If Linux hangs, nothing on Linux can be trusted to notice
      \end{itemize}
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \slideimg[0.36\textheight]{\linewidth}{V02}{Late result under load}\\[3pt]
      \slideimg[0.2\textheight]{0.48\linewidth}{V03}{Delay invisible to receiver}\hfill
      \slideimg[0.2\textheight]{0.48\linewidth}{V04}{Linux hang}
    \end{column}
  \end{columns}
  \note{Example: the Pi runs a neural network every 100 ms and sends ``obstacle / no
  obstacle'' to a microcontroller that drives a motor. A late or frozen answer is a safety
  problem, not just a performance one.}
\end{frame}

\begin{frame}{We ask three measurable questions, using only stock OS mechanisms}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \begin{enumerate}
        \setlength{\itemsep}{6pt}
        \item How much does Linux \textbf{resource contention} delay a time-sensitive data path?
        \item Which \textbf{stock OS mechanisms} recover that delay, and which cannot?
        \item Can an \textbf{independent real-time supervisor} bound the reaction to Linux failures?
      \end{enumerate}
      \vspace{6pt}
      \muted{\footnotesize No new OS, no kernel patches: we combine, configure and measure.}
    \end{column}
    \begin{column}{0.43\textwidth}
      \centering\slideimg[0.62\textheight]{\linewidth}{V07}{Three research questions}
    \end{column}
  \end{columns}
  \note{We do not write a new OS or patch the kernel; we combine and configure existing OS
  mechanisms and measure the effect.}
\end{frame}

\section{Background}

\begin{frame}{The pattern is known; its failsafe latency on commodity Linux is not measured}
  \begin{columns}[T]
    \begin{column}{0.64\textwidth}
      \footnotesize
      \begin{itemize}
        \setlength{\itemsep}{3pt}
        \item \textbf{System-Level Simplex} \cite{bak2009}: untrusted complex subsystem + simple
              safety subsystem on separate hardware
        \item \textbf{Age of Information} \cite{kaul2012queues,kaul2012infocom}: newest-first
              delivery keeps data freshest
        \item \textbf{Multicore interference} \cite{yun2013memguard,bechtel2018deeppicar}: dedicated
              cores do not isolate memory bandwidth
        \item \textbf{Pi~5 + PREEMPT\_RT} \cite{arxiv2604}: SCHED\_FIFO 99 worst case
              \SI{1.8}{ms} stock vs \SI{0.22}{ms} RT kernel, stressed \evid{PUBLISHED}
      \end{itemize}
      \vspace{4pt}
      \begin{block}{Gap}
        \footnotesize Simplex papers report no failsafe-latency numbers. We measure them end to end
        on Linux + FreeRTOS.
      \end{block}
    \end{column}
    \begin{column}{0.34\textwidth}
      \centering\slideimg[0.66\textheight]{\linewidth}{V06}{Related work map}
    \end{column}
  \end{columns}
  \note{Sources in docs/research/RELATED\_WORK.md. Say clearly that these are their results,
  not ours. The 1.8 / 0.22 ms figures are Table 4 of arXiv:2604.19275 (4 ms loop, stressed).}
\end{frame}

\begin{frame}{CPU scheduling decides who runs, not who uses memory bandwidth}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{5pt}
        \item 4 $\times$ Cortex-A76, \SI{512}{KB} private L2 each
        \item \SI{2}{MB} shared L3, shared LPDDR4X
        \item Pinning a core isolates CPU time,\\ \alert{not} L3 or DRAM bandwidth
      \end{itemize}
      \vspace{4pt}
      \muted{\footnotesize Leads to hypothesis H2.}
    \end{column}
    \begin{column}{0.54\textwidth}
      \centering\slideimg[0.66\textheight]{\linewidth}{V05}{BCM2712: private L2, shared L3 and DRAM}
    \end{column}
  \end{columns}
  \note{This is why we expect tuning to fix CPU contention but not memory contention
  (hypothesis H2).}
\end{frame}

\section{System design}

\begin{frame}{Linux proposes; the STM32 has final authority over the output}
  \begin{columns}[c]

      \end{itemize}
      \vspace{3pt}
      \textbf{\color{teal}Real-time domain} \muted{(STM32F446RE, FreeRTOS)}
      \begin{itemize}
        \item UART RX, supervisor, status, failsafe output
      \end{itemize}
      \vspace{3pt}
      One UART link + one GPIO sync line

\begin{frame}{Freshness is measured as Age of Information at the MCU}
  \begin{columns}[c]
    \begin{column}{0.40\textwidth}
      \footnotesize
      capture $\rightarrow$ inference $\rightarrow$ buffer $\rightarrow$ UART $\rightarrow$ supervisor $\rightarrow$ action
      \vspace{6pt}
      \[ \Delta(t) = t - u(t), \quad u(t)=\max\{g_i : r_i \le t\} \]
      \muted{$g_i$: input time, $r_i$: arrival at MCU}
      \vspace{4pt}
      \begin{itemize}
        \item grows linearly between deliveries
        \item drops at each fresh delivery
      \end{itemize}
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \slideimg[0.36\textheight]{\linewidth}{V10}{End-to-end data path}\\[3pt]
      \slideimg[0.3\textheight]{\linewidth}{V20}{AoI sawtooth}
    \end{column}
  \end{columns}
  \note{AoI grows linearly between deliveries and drops at each fresh delivery. Average AoI and
  the fraction of time above a threshold are our freshness metrics. Definition follows Kaul et al.}
\end{frame}

\begin{frame}{A latest-value buffer drops stale results instead of queueing them}
  \begin{columns}[c]
    \begin{column}{0.40\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{6pt}
        \item \textbf{FIFO queue}: under overload, results wait and arrive old
        \item \textbf{Latest value}: an unsent old result is overwritten
        \item Theory \cite{kaul2012queues}: lower average age; \alert{no hard bound}
      \end{itemize}
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering\slideimg[0.7\textheight]{\linewidth}{V14}{FIFO queue vs 1-slot latest value}
    \end{column}
  \end{columns}
  \note{Theory predicts lower average age, but no hard bound; the queueing results assume
  memoryless arrivals and service. We measure how much it helps on real hardware (E3).}
\end{frame}

\begin{frame}{Two clocks are mapped by a T1--T4 exchange and checked by a GPIO edge}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \centering
      \begin{tikzpicture}[font=\footnotesize]
        \draw[navy, thick] (0,0) node[above, align=center] {\textbf{Pi~5}\\ \scriptsize CLOCK\_MONOTONIC\_RAW} -- (0,-3.4);
        \draw[teal, thick] (4,0) node[above, align=center] {\textbf{STM32}\\ \scriptsize timer, \si{\micro s}} -- (4,-3.4);
        \draw[flow] (0,-0.5) node[left] {$T_1$} -- node[lbl, above, sloped] {ECHO\_REQ} (4,-1.3) node[right] {$T_2$};
        \draw[flow, teal] (4,-1.9) node[right] {$T_3$} -- node[lbl, above, sloped] {ECHO\_RESP} (0,-2.8) node[left] {$T_4$};
        \foreach \y in {-0.5,-2.8} \fill[navy] (0,\y) circle (1.5pt);
        \foreach \y in {-1.3,-1.9} \fill[teal] (4,\y) circle (1.5pt);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize
      \[ \theta = \frac{(T_2 - T_1) + (T_3 - T_4)}{2} \]
      \[ \delta = (T_4 - T_1) - (T_3 - T_2) \]
      \muted{$\theta$: clock offset, $\delta$: round-trip link delay;\\ assumes symmetric delay (as in NTP)}
      \vspace{4pt}
      \begin{itemize}
        \item Drift: linear fit over many exchanges, min-delay filter
        \item Ground truth: GPIO17 edge $\rightarrow$ PA0 input capture
        \item Target: sub-millisecond \evid{to be measured}
      \end{itemize}
    \end{column}
  \end{columns}
  \note{Never subtract timestamps from different clocks without this mapping. Asymmetric UART
  delay biases theta by half the asymmetry; the GPIO edge checks this independently. Image V19 can
  go in backup.}
\end{frame}

\begin{frame}{Hypotheses: predictions to be tested, not results}
  \begin{enumerate}
    \setlength{\itemsep}{7pt}
    \item[\textbf{H1}] Contention inflates P99 far more than P50
    \item[\textbf{H2}] Tuning recovers CPU-contention tails, \alert{not} memory-contention tails
    \item[\textbf{H3}] Latest value lowers average AoI and violation time (no hard bound)
    \item[\textbf{H4}] STM32 failsafe latency stays within its budget for every fault type
  \end{enumerate}
  \vspace{8pt}
  \muted{\footnotesize Each hypothesis maps to one experiment: H1--E1, H2--E2, H3--E3, H4--E4.}
  \note{These are predictions; results slides come after the experiments. H2 is supported by prior
  work (MemGuard, DeepPicar) on other platforms, not yet on the Pi~5.}
\end{frame}
