\section{Cơ sở lý thuyết}
\begin{frame}{2.1 Kiến trúc Simplex}
\begin{itemize}
\item \textbf{Simplex} là kiến trúc an toàn chia hệ thống thành hai phần chạy song song \cite{bak2009}:
\begin{itemize}
\item Thành phần phức tạp (complex subsystem): hiệu năng cao nhưng khó kiểm chứng, ví dụ mạng nơ-ron chạy trên Linux.
\item Bộ giám sát an toàn (safety supervisor): đơn giản, kiểm chứng được và luôn giữ quyền quyết định cuối cùng.
\end{itemize}
\end{itemize}\medskip
Hình~\ref{fig:deck-8} cho thấy đề xuất đi qua bộ giám sát trước khi tới đầu ra.\par\medskip
\fig[.40\textheight]{V06}{\label{fig:deck-8}Simplex: bộ giám sát quyết định đầu ra.}
\end{frame}
\begin{frame}{2.1 Kiến trúc Simplex (tiếp)}
\begin{itemize}\item Biến thể \textbf{System-Level Simplex} đặt hai phần trên hai phần cứng riêng, nên khi thành phần phức tạp bị lỗi hoặc treo, bộ giám sát vẫn hoạt động.
\item Trong đồ án: Raspberry Pi 5 là thành phần phức tạp, STM32 là bộ giám sát.

\begin{frame}{2.2 Lập lịch thời gian thực trên Linux (tiếp): PREEMPT\_RT}
\normalsize
\begin{itemize}\item \textbf{Kết quả công bố \cite{arxiv2604}:} Kernel PREEMPT\_RT giảm các đoạn kernel không thể ngắt: trên Raspberry Pi 5 dưới tải nặng, độ trễ xấu nhất giảm từ 1,8 ms xuống 0,22 ms \cite{arxiv2604}.
\end{itemize}
\begin{itemize}
\item Đây là kết quả công bố của tác giả \cite{arxiv2604}, không phải số đo của nhóm; giá trị gốc là 1\,848 và 224 $\mu$s, SCHED\_FIFO ưu tiên 99.
\item PREEMPT\_RT được thử bằng kernel Real-time Ubuntu đóng gói sẵn (P3), không sửa hoặc tự biên dịch kernel.
\item Lịch đều ở Hình~\ref{fig:deck-10} chỉ minh họa tác động mong muốn; độ trễ đuôi và deadline vẫn phải đo dưới tải.
\end{itemize}
\note{V4 ghi \cite{bechtel2018deeppicar} cho PREEMPT\_RT nhưng nguồn này là \cite{arxiv2604} trong danh mục hiện có; sửa số trích dẫn, giữ nguyên câu.}

CI (Continuous Integration) và script cài đặt Raspberry Pi & Hoàn thành, chưa chạy trên phần cứng\\
Firmware, bridge, thí nghiệm & Chưa thực hiện\\
\bottomrule
\end{tabular}\par

\note{Tiến độ theo tài liệu dự án ngày 06/10/2026; không chạy lại kiểm thử giao thức trong lần sửa slide. Cross-compile và CI không thay cho kiểm thử Raspberry Pi hoặc STM32 thật.}
\vfill
\begin{itemize}
\item Các kết quả hiện tại xác nhận thiết kế và phần mềm trên máy tính; chưa có số đo Raspberry Pi 5 hoặc STM32.
\item Điểm chuyển tiếp: nâng cấp bộ giám sát hai bộ định thời, giữ chốt FAILSAFE và kiểm chứng bằng đường UART thật.
\end{itemize}
\end{frame}

\section{Kết luận}
\begin{frame}{6.1 Kết luận}
\begin{itemize}
\item AI trên thiết bị biên ngày càng điều khiển trực tiếp hệ thống vật lý, nên kết quả phải vừa đúng vừa kịp thời; Linux đơn thuần tối ưu thông lượng và không bảo đảm điều này.
\item Đồ án đề xuất (Hình~\ref{fig:deck-16}) Linux + FreeRTOS theo Simplex, với latest-value buffer, bộ giám sát hai bộ định thời và đồng bộ đồng hồ; dùng cơ chế sẵn có của hệ điều hành.
\end{itemize}
\vfill
\fig[.48\textheight]{V08}{\label{fig:deck-16}Linux đề xuất; STM32 quyết định đầu ra.}
\end{frame}
\begin{frame}{6.1 Kết luận (tiếp)}
\begin{itemize}
\item Ý nghĩa: hướng kiến trúc này phù hợp xu hướng SoC tích hợp nhân ứng dụng và nhân thời gian thực; các phép đo sẽ chỉ ra cơ chế nào thực sự cần thiết và chi phí của chúng, làm cơ sở thiết kế cho robot, drone và máy công nghiệp dùng AI.
\item Hiện trạng (Hình~\ref{fig:next-stage}): đã hoàn thành thiết kế, thư viện giao thức và kế hoạch thí nghiệm; bước tiếp theo là triển khai trên phần cứng để trả lời RQ1–RQ3 bằng số liệu đo.
\end{itemize}
