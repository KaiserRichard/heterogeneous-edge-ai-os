\begin{frame}{1.1 Động lực nghiên cứu}
\relax
\small

\section{Cơ sở lý thuyết}
\begin{frame}{2.1 Kiến trúc Simplex}
\relax
\small
\setlength{\parskip}{0pt}
\begin{itemize}\setlength{\itemsep}{0pt}\setlength{\parsep}{0pt}\setlength{\parskip}{0pt}
\item Hệ thống gồm một thành phần phức tạp, hiệu năng cao và một thành phần giám sát đơn giản, đáng tin cậy, chạy trên phần cứng riêng \cite{bak2009}.
\item Bộ giám sát có quyền quyết định cuối cùng; đầu ra của thành phần phức tạp chỉ là đề xuất.
\item Trong đồ án: Raspberry Pi 5 là thành phần phức tạp, STM32 là bộ giám sát.
\end{itemize}
\vspace{-12pt}
\centering\slideimg[0.62\textheight]{0.45\textwidth}{V06}{Kiến trúc Simplex}\figcap{Kiến trúc Simplex}

\end{frame}
\begin{frame}{2.2 Lập lịch thời gian thực trên Linux}
\relax
\small
\setlength{\parskip}{0pt}
\begin{itemize}\setlength{\itemsep}{0pt}\setlength{\parsep}{0pt}\setlength{\parskip}{0pt}
\item Bộ lập lịch mặc định của Linux chia sẻ CPU công bằng nên tác vụ quan trọng có thể phải chờ.
\item SCHED\_FIFO cho tác vụ ưu tiên cố định; ghim CPU (CPU affinity) và cgroups giới hạn tác vụ nền.
\item Kernel PREEMPT\_RT giảm các đoạn không thể ngắt trong kernel. Trên Raspberry Pi 5 dưới tải nặng, độ trễ xấu nhất của SCHED\_FIFO giảm từ 1,8 ms xuống 0,22 ms \cite{arxiv2604}.
\end{itemize}
\vspace{-13pt}
\centering\slideimg[0.62\textheight]{0.45\textwidth}{V11}{Lập lịch mặc định và sau tinh chỉnh}\figcap{Lập lịch mặc định và sau tinh chỉnh}

\note{Số công bố làm tròn từ 1848 và 224 microsecond ở ưu tiên 99; không phải số đo của nhóm hay cấu hình P1 ưu tiên 50. Lịch đều trong ảnh chỉ là minh họa.}
\end{frame}
\begin{frame}{2.3 Tranh chấp tài nguyên trên vi xử lý đa nhân}
\relax
\small
\setlength{\parskip}{0pt}
\begin{itemize}\setlength{\itemsep}{0pt}\setlength{\parsep}{0pt}\setlength{\parskip}{0pt}
\item Raspberry Pi 5 có 4 nhân Cortex-A76 dùng chung bộ nhớ đệm L3 (Level 3) 2 MB và bộ nhớ LPDDR4X (Low-Power Double Data Rate 4X).
\item Ghim CPU quyết định tác vụ chạy ở nhân nào, nhưng không cô lập được băng thông bộ nhớ \cite{yun2013memguard}.
\item Tác vụ AI trên CPU bị chậm đáng kể khi nhân khác tranh chấp bộ nhớ \cite{bechtel2018deeppicar}.
\end{itemize}
\vspace{-12pt}
\centering\slideimg[0.62\textheight]{0.45\textwidth}{V05}{Các nhân dùng chung cache và băng thông bộ nhớ}\figcap{Các nhân dùng chung cache và băng thông bộ nhớ}

\note{Ảnh dùng mô hình bộ nhớ trừu tượng; không mô tả cách đóng gói LPDDR4X trên bo Pi. Kết quả nghiên cứu trước không phải benchmark Pi của nhóm.}
\end{frame}
\begin{frame}{2.4 Tuổi thông tin (Age of Information)}
\relax
\small
\setlength{\parskip}{0pt}
\begin{itemize}\setlength{\itemsep}{0pt}\setlength{\parsep}{0pt}\setlength{\parskip}{0pt}
\item AoI (Age of Information) tại thời điểm $t$: $\Delta(t)=t-u(t)$, với $u(t)$ là thời điểm lấy mẫu của dữ liệu mới nhất đã nhận \cite{kaul2012infocom}.
\item AoI tăng tuyến tính giữa hai lần nhận và giảm khi có dữ liệu mới.
\item Chỉ số đánh giá: AoI trung bình, AoI đỉnh, tỉ lệ thời gian AoI vượt ngưỡng $\tau$.
\item Ưu tiên dữ liệu mới nhất giúp giảm AoI trung bình so với hàng đợi FIFO (First-In, First-Out) \cite{kaul2012queues}.
\end{itemize}
\vspace{-12pt}
\centering\slideimg[0.62\textheight]{0.45\textwidth}{V20}{Tuổi thông tin theo thời gian (minh họa)}\figcap{Tuổi thông tin theo thời gian (minh họa)}

\note{Kết luận lý thuyết phụ thuộc mô hình hàng đợi. Latest-value buffer chỉ thay kết quả chưa gửi, không hủy suy luận hoặc frame đang truyền; không tạo cận trên AoI cứng.}
\end{frame}

\end{frame}
\begin{frame}{3.8 Vì sao cần hai bộ định thời}

Bridge vẫn gửi heartbeat nhưng có thể chuyển tiếp kết quả cũ, nên chỉ heartbeat là không đủ.\par

\begin{frame}{4.2 Các thí nghiệm}

\begin{frame}{4.3 Đo phản ứng khi có lỗi}
\relax
\small
\setlength{\parskip}{0pt}

\item Đo thời điểm lỗi $t_f$, quyết định $t_d$ và đầu ra an toàn $t_s$ trên cùng một logic analyzer.
\item Ngân sách dự kiến: timeout + chu kỳ supervisor + trễ điều phối + thực thi + trễ đầu ra.
\item Log trên Linux không đo được lúc Linux treo, nên phải đo từ bên ngoài.
\end{itemize}
