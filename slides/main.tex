\begin{frame}{Nghiên cứu trước định hướng cách giám sát và đo độ mới}
  \relax % Ngăn Beamer đọc nhóm nội dung đầu tiên thành framesubtitle.
  \begin{columns}[T]
    \begin{column}{0.66\textwidth}
      {\small
      \textbf{Simplex:} tách hệ phức tạp khỏi bộ giám sát tin cậy trên phần cứng riêng \cite{bak2009}.\par\vspace{4pt}
      \textbf{Age of Information (AoI):} tuổi dữ liệu mới nhất tại nơi nhận; khác độ trễ riêng gói đã giao \cite{kaul2012queues,kaul2012infocom}.\par\vspace{4pt}
      \textbf{PREEMPT\_RT:} Preemption for Real-Time (RT); kernel hỗ trợ preemption, giảm một số độ trễ chờ lập lịch.\par}
      \vspace{4pt}
      {\footnotesize SCHED\_FIFO (Scheduling – First-In, First-Out), viết tắt FIFO. Với ưu tiên 99, cực đại công bố khi có tải: 1,8 ms stock $\rightarrow$ 0,22 ms RT \published\ \cite{arxiv2604}.\par}
    \end{column}
    \begin{column}{0.32\textwidth}
      \centering\slideimg[0.53\textheight]{\linewidth}{V06}{Related work map}
    \end{column}
  \end{columns}
  \src{Khoảng trống đồ án: đo từ lỗi đến đầu ra failsafe trên Linux + FreeRTOS. Nguồn: R1–R3.}
  \note{R1 không tìm thấy phân bố định lượng fault-to-output trong bài Simplex đã đọc; không suy rộng thành cả lĩnh vực thiếu kết quả. Nhóm áp dụng giám sát lấy cảm hứng từ Simplex, không có bộ điều khiển dự phòng đã chứng minh ổn định. Số 1,8/0,22 ms làm tròn từ bảng 4 của bài arXiv, cấu hình FIFO 99 [CÔNG BỐ]. Bài dùng nền tảng và RT budget khác nhóm, không đo UART; không dự báo AoI hoặc failsafe.}
\end{frame}
